#Requires -Version 5.1
# Access renders validated release metadata; invoke with native PowerShell.
Set-PSDebug -Off
$ErrorActionPreference = 'Stop'
$ProgressPreference = 'SilentlyContinue'
$VerbosePreference = 'SilentlyContinue'
$DebugPreference = 'SilentlyContinue'
$version = '__ONECREW_VERSION__'
$bootstrapUpdate = '__ONECREW_BOOTSTRAP_UPDATE__'
$url = '__ONECREW_WINDOWS_URL__'
$sha = '__ONECREW_WINDOWS_SHA256__'
$size = '__ONECREW_WINDOWS_SIZE__'
$stage = $null
$cliInstalled = $false
$downloadFailure = $null
$phase = 'preflight'
$failureCode = 'installation_failed'
$nativeExitCode = $null
# Start-Process redirects native streams without PowerShell 5.1 converting stderr
# into terminating errors. Wait only for this command, not its service descendants.
function Invoke-OneCrewCommand {
    param([string]$Executable, [string]$CommandArguments, [int]$TimeoutSeconds)
    $outputPath = Join-Path $stage 'command.stdout'
    $errorPath = Join-Path $stage 'command.stderr'
    $process = $null
    try {
        $process = Start-Process -FilePath $Executable -ArgumentList $CommandArguments -NoNewWindow -PassThru -RedirectStandardOutput $outputPath -RedirectStandardError $errorPath -ErrorAction Stop
        # Retain a handle so ExitCode is available even when the command exits quickly.
        $null = $process.Handle
        try {
            $process | Wait-Process -Timeout $TimeoutSeconds -ErrorAction Stop
        } catch {
            if (-not $process.HasExited) {
                # Stop only the command started here; never kill all OneCrew processes.
                $process | Stop-Process -ErrorAction SilentlyContinue
                throw 'command_timeout'
            }
        }
        $process.WaitForExit() # Flush redirected output after this process has exited.
        $process.Refresh()
        return [PSCustomObject]@{
            ExitCode = $process.ExitCode
            Stdout = [IO.File]::ReadAllText($outputPath)
            Stderr = [IO.File]::ReadAllText($errorPath)
        }
    } finally {
        if ($process) { $process.Dispose() }
    }
}
function Write-ConnectionGuidance {
    [Console]::Error.WriteLine('CLI installation complete. Use installed_path and follow the installation guide for your AI client; returned connection options are alternatives, not steps to run in full.')
}
function Confirm-OneCrewService {
    $deadline = [DateTime]::UtcNow.AddSeconds(60)
    $script:phase = 'service_verification'
    $script:failureCode = 'service_verification_timeout'
    while ([DateTime]::UtcNow -lt $deadline) {
        $status = $null
        $remaining = [Math]::Max(1, [int][Math]::Ceiling(($deadline - [DateTime]::UtcNow).TotalSeconds))
        # status may start the service; the command itself has a bounded wait.
        $probe = Invoke-OneCrewCommand $installed 'status --json' $remaining
        $script:nativeExitCode = $probe.ExitCode
        try {
            if ($probe.ExitCode -eq 0) { $status = $probe.Stdout | ConvertFrom-Json }
        } catch { $status = $null }
        if ($status -and $status.running -eq $true) {
            if ($status.version -ceq $version -and $status.binary_sha256 -ceq $sha) {
                if ($status.startup.phase -ceq 'ready' -and $status.data.state -ceq 'ready') {
                    [Console]::Error.WriteLine('OneCrew service verified: expected version and binary, startup and data ready.')
                    return
                }
                if ($status.startup.phase -ceq 'failed') { $script:failureCode = 'service_startup_failed'; throw 'service verification' }
            } else {
                # Recheck after stop; acknowledgement does not prove ownership changed.
                $remaining = [int][Math]::Ceiling(($deadline - [DateTime]::UtcNow).TotalSeconds)
                if ($remaining -le 0) { break }
                $null = Invoke-OneCrewCommand $installed 'stop --json' $remaining
            }
        }
        Start-Sleep -Seconds 1
    }
    throw 'service verification timeout'
}
try {
    if ([Environment]::OSVersion.Platform -ne [PlatformID]::Win32NT) { throw 'platform' }
    $nativeArch = $env:PROCESSOR_ARCHITEW6432
    if (-not $nativeArch) { $nativeArch = $env:PROCESSOR_ARCHITECTURE }
    # OSArchitecture identifies ARM even when PowerShell runs under x64 emulation.
    if ('System.Runtime.InteropServices.RuntimeInformation' -as [type]) {
        $nativeArch = [System.Runtime.InteropServices.RuntimeInformation]::OSArchitecture.ToString()
    }
    if (-not [Environment]::Is64BitOperatingSystem -or $nativeArch -notin @('AMD64', 'X64')) { throw 'architecture' }
    $uri = [Uri]$url
    if ($uri.Scheme -ne 'https' -or $uri.UserInfo -or $uri.Fragment) { throw 'url' }
    if ($sha -cnotmatch '^[0-9a-f]{64}$' -or $size -notmatch '^[1-9][0-9]*$') { throw 'metadata' }
    if ($bootstrapUpdate -cnotmatch '^[01]$') { throw 'bootstrap capability' }
    if (-not $env:LOCALAPPDATA) { throw 'localappdata' }
    $dest = Join-Path $env:LOCALAPPDATA 'OneCrew\bin'
    $installed = Join-Path $dest 'onecrew.exe'
    [IO.Directory]::CreateDirectory($dest) | Out-Null
    $stage = Join-Path $dest ('.onecrew-install.' + [Guid]::NewGuid().ToString('N'))
    [IO.Directory]::CreateDirectory($stage) | Out-Null
    $candidate = Join-Path $stage 'onecrew.exe'
    $phase = 'download'
    $failureCode = 'download_failed'
    # No redirect may forward the scoped ticket; never render exception details.
    try {
        Invoke-WebRequest -UseBasicParsing -Uri $uri -OutFile $candidate -MaximumRedirection 0 -TimeoutSec 600 -ErrorAction Stop | Out-Null
    } catch {
        $httpStatus = 0
        if ($_.Exception.Response) { $httpStatus = [int]$_.Exception.Response.StatusCode }
        if ($httpStatus -in @(401, 403)) {
            $failureCode = 'download_access_denied'
            $downloadFailure = 'download_access_denied: Return to Access to check your account and installation access.'
        } elseif ($httpStatus -in @(404, 409, 410)) {
            $failureCode = 'download_refresh_required'
            $downloadFailure = 'download_refresh_required: Reopen the original installation link and download the new script. If that link has expired, get a new prompt from Access.'
        } elseif ($httpStatus -eq 0 -or $httpStatus -eq 429 -or ($httpStatus -ge 500 -and $httpStatus -le 599)) {
            $failureCode = 'download_retryable'
            $downloadFailure = 'download_retryable: Check the network or wait before retrying. Reopen the original installation link if its download URLs have expired.'
        }
        throw 'download'
    }
    $phase = 'binary_validation'
    $failureCode = 'binary_validation_failed'
    if ((Get-Item -LiteralPath $candidate).Length -ne [long]$size) { throw 'size' }
    if ((Get-FileHash -LiteralPath $candidate -Algorithm SHA256).Hash.ToLowerInvariant() -cne $sha) { throw 'hash' }
    # Verify PE machine type before executing the authenticated candidate.
    $stream = [IO.File]::OpenRead($candidate)
    $reader = [IO.BinaryReader]::new($stream)
    try {
        if ($reader.ReadUInt16() -ne 0x5a4d) { throw 'PE' }
        $stream.Position = 0x3c
        $peOffset = $reader.ReadUInt32()
        if ($peOffset -gt $stream.Length - 6) { throw 'PE' }
        $stream.Position = $peOffset
        if ($reader.ReadUInt32() -ne 0x4550 -or $reader.ReadUInt16() -ne 0x8664) { throw 'PE architecture' }
    } finally { $reader.Dispose() }
    $reported = Invoke-OneCrewCommand $candidate '--version' 30
    $nativeExitCode = $reported.ExitCode
    if ($reported.ExitCode -ne 0 -or $reported.Stdout.TrimEnd([char[]]"`r`n") -cne "onecrew $version") { throw 'version' }
    if ($bootstrapUpdate -eq '1') {
        # Execute the authenticated candidate, never a possibly older installed CLI.
        $phase = 'bootstrap'
        $failureCode = 'bootstrap_failed'
        $nativeExitCode = $null
        $cliInstalled = $null # Until the command returns, replacement state is unknown.
        # Start-Process joins ArgumentList strings; quote the path inside one string.
        $arguments = '__bootstrap-update --install-path "{0}" --expected-version {1} --expected-sha256 {2} --expected-size {3} --json' -f $installed, $version, $sha, $size
        $result = Invoke-OneCrewCommand $candidate $arguments 120
        $nativeExitCode = $result.ExitCode
        if ($result.ExitCode -ne 0) {
            if ($result.Stderr -match '(?m)^onecrew: ([a-z][a-z0-9_]{0,79})\r?$') { $failureCode = $Matches[1] }
            throw 'bootstrap'
        }
        $cliInstalled = $true
        $phase = 'installer_result'
        $failureCode = 'invalid_installer_result'
        $outcome = $result.Stdout | ConvertFrom-Json
        switch -CaseSensitive ($outcome.action) {
            'installed' {} # First installation returns login guidance before service startup.
            'updated' { Confirm-OneCrewService }
            'already_current' { Confirm-OneCrewService }
            default { throw 'installer result' }
        }
        [Console]::Out.Write($result.Stdout)
        Write-ConnectionGuidance
        exit 0
    }
    $phase = 'binary_replacement'
    $failureCode = 'binary_replacement_failed'
    $cliInstalled = $null
    if ([IO.File]::Exists($installed)) {
        [IO.File]::Replace($candidate, $installed, $null)
    } else {
        [IO.File]::Move($candidate, $installed)
    }
    $cliInstalled = $true
    Write-Output 'action=installed'
    Write-Output ('installed_path="' + $installed + '"')
    Write-Output ('next_action="& ''' + $installed.Replace("'", "''") + ''' login --json"')
    Write-Output 'Complete authorization in the browser opened by the login command.'
    Write-Output 'Wait for the login command to finish and return its final JSON before continuing.'
    Write-ConnectionGuidance
} catch {
    if ($_.Exception.Message -eq 'command_timeout') { $failureCode = 'command_timeout'; $nativeExitCode = $null }
    # Fixed fields only: never print exception details, download URLs or service tokens.
    [Console]::Error.WriteLine((@{ stage = $phase; error = $failureCode; native_exit_code = $nativeExitCode; cli_installed = $cliInstalled } | ConvertTo-Json -Compress))
    if ($downloadFailure) {
        [Console]::Error.WriteLine($downloadFailure)
    } elseif ($cliInstalled -and $phase -eq 'service_verification') {
        [Console]::Error.WriteLine('OneCrew CLI is installed, but service verification failed. Report the failed stage and error; do not repeat an unbounded status command.')
    } else {
        [Console]::Error.WriteLine('OneCrew installation did not complete. Keep the current files and report the failed stage; do not clear the installation directory or manually replace the executable.')
    }
    if ($null -ne $nativeExitCode -and $nativeExitCode -ne 0) { exit $nativeExitCode }
    exit 1
} finally {
    if ($stage -and [IO.Directory]::Exists($stage)) {
        Remove-Item -LiteralPath $stage -Recurse -Force -ErrorAction SilentlyContinue
    }
}
