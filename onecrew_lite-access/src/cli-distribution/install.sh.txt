#!/bin/sh
# Access renders validated release metadata into these single-quoted values.
set +x
set -eu
umask 077
version='__ONECREW_VERSION__'
bootstrap_update='__ONECREW_BOOTSTRAP_UPDATE__'
stage=''
cleanup() { if [ -n "$stage" ]; then rm -rf "$stage"; fi; }
fail() { printf '%s\n' 'OneCrew installation failed; existing installation preserved. Retry from Access.' >&2; exit 1; }
download_fail() {
  case "$1" in
    401|403) message='download_access_denied: Return to Access to check your account and installation access.' ;;
    404|409|410) message='download_refresh_required: Reopen the original installation link and download the new script. If that link has expired, get a new prompt from Access.' ;;
    000|429|5??) message='download_retryable: Check the network or wait before retrying. Reopen the original installation link if its download URLs have expired.' ;;
    *) fail ;;
  esac
  printf '%s\n' "$message" >&2
  exit 1
}
connection_guidance() { printf '%s\n' 'CLI installation complete. Use installed_path and follow the installation guide for your AI client; returned connection options are alternatives, not steps to run in full.' >&2; }
service_fail() { printf '%s\n' 'OneCrew CLI is installed, but service verification failed. Run the installed CLI with status --json to inspect recovery guidance.' >&2; exit 1; }
status_field() { /usr/bin/plutil -extract "$1" raw -o - "$stage/service.json" 2>/dev/null || :; }
verify_service() {
  deadline=$(( $(date +%s) + 60 ))
  while [ "$(date +%s)" -lt "$deadline" ]; do
    # status starts the installed binary's service when none is reachable.
    # Keep its JSON private; never launch web or print a one-use URL.
    if "$dest/onecrew" status --json > "$stage/service.json" 2>/dev/null; then
      if [ "$(status_field running)" = true ]; then
        if [ "$(status_field version)" = "$version" ] && [ "$(status_field binary_sha256)" = "$sha" ]; then
          if [ "$(status_field startup.phase)" = ready ] && [ "$(status_field data.state)" = ready ]; then
            printf '%s\n' 'OneCrew service verified: expected version and binary, startup and data ready.' >&2
            return 0
          fi
          [ "$(status_field startup.phase)" != failed ] || service_fail
        else
          # A stop acknowledgement is not completion. Recheck until the expected
          # service owns the data directory; an old plugin can race this restart.
          "$dest/onecrew" stop --json > "$stage/stop.json" 2>/dev/null || :
        fi
      fi
    fi
    sleep 1
  done
  service_fail
}
trap cleanup EXIT
trap 'exit 1' HUP INT TERM

[ "$(uname -s)" = Darwin ] || fail
arch=$(uname -m)
# A shell running under Rosetta reports x86_64 on an Apple Silicon Mac.
if [ "$arch" = x86_64 ] && [ "$(sysctl -in sysctl.proc_translated 2>/dev/null || :)" = 1 ]; then arch=arm64; fi
case "$arch" in
  arm64)
    url='__ONECREW_MAC_ARM_URL__'
    sha='__ONECREW_MAC_ARM_SHA256__'
    size='__ONECREW_MAC_ARM_SIZE__'
    cpu='0c000001' ;;
  x86_64)
    url='__ONECREW_MAC_INTEL_URL__'
    sha='__ONECREW_MAC_INTEL_SHA256__'
    size='__ONECREW_MAC_INTEL_SIZE__'
    cpu='07000001' ;;
  *) fail ;;
esac
case "$url" in https://*) ;; *) fail ;; esac
case "$size" in ''|*[!0-9]*) fail ;; esac
case "$sha" in *[!0-9a-f]*|'') fail ;; esac
[ "${#sha}" = 64 ] || fail
case "$bootstrap_update" in 0|1) ;; *) fail ;; esac
[ -n "${HOME:-}" ] || fail
dest="$HOME/.local/bin"
mkdir -p "$dest" 2>/dev/null || fail
stage=$(mktemp -d "$dest/.onecrew-install.XXXXXXXX") || fail
candidate="$stage/onecrew"
# Ignore ~/.curlrc. No redirects, verbose errors, or response bodies reach output.
status=$(curl -q --silent --fail --proto '=https' --proto-redir '=https' --max-redirs 0 --connect-timeout 30 --max-time 600 --output "$candidate" --write-out '%{http_code}' "$url" 2>/dev/null) || download_fail "${status:-000}"
[ "$status" = 200 ] || download_fail "$status"
[ "$(stat -f '%z' "$candidate" 2>/dev/null)" = "$size" ] || fail
actual=$(shasum -a 256 "$candidate" 2>/dev/null) || fail
[ "${actual%% *}" = "$sha" ] || fail
header=$(od -An -tx1 -N8 "$candidate" 2>/dev/null | tr -d ' \n')
[ "$header" = "cffaedfe$cpu" ] || fail
chmod 755 "$candidate" 2>/dev/null || fail
reported=$("$candidate" --version 2>/dev/null) || fail
[ "$reported" = "onecrew $version" ] || fail
if [ "$bootstrap_update" = 1 ]; then
  # The verified new binary owns drain/lease/replace. The installed .21 need not know update.
  "$candidate" __bootstrap-update --install-path "$dest/onecrew" --expected-version "$version" --expected-sha256 "$sha" --expected-size "$size" --json > "$stage/update.json" || exit 1
  action=$(/usr/bin/plutil -extract action raw -o - "$stage/update.json" 2>/dev/null) || service_fail
  case "$action" in
    installed) ;; # First installation returns login guidance before service startup.
    updated|already_current) verify_service ;;
    *) service_fail ;;
  esac
  cat "$stage/update.json"
  connection_guidance
  exit 0
fi
# Stage and destination share a filesystem; rename replaces atomically.
[ ! -d "$dest/onecrew" ] || fail
mv -f "$candidate" "$dest/onecrew" 2>/dev/null || fail
printf 'action=installed\ninstalled_path="%s"\n' "$dest/onecrew"
# POSIX single-quote escaping keeps the returned command runnable for spaces/apostrophes.
quoted=$(printf '%s' "$dest/onecrew" | sed "s/'/'\\\\''/g")
printf "next_action=\"'%s' login --json\"\n" "$quoted"
printf '%s\n' 'Complete authorization in the browser opened by the login command.' 'Wait for the login command to finish and return its final JSON before continuing.'
connection_guidance
